```javascript
// Central Neural AI Lumen (Icosahedral Core)
const lumenGeo = new THREE.IcosahedronGeometry(1.0, 1);
const lumenMesh = new THREE.Mesh(lumenGeo, ivoryCoreMat);
coreGroup.add(lumenMesh);
coreGroup.add(new THREE.PointLight(0xf1e0d0, 2.5, 18));

// 1300s Equator Astrolabe Ring (Gilded Brass)
const ringAGeo = new THREE.TorusGeometry(3.6, 0.05, 24, 160);
const ringAMesh = new THREE.Mesh(ringAGeo, antiqueBrassMat);

// Tilted Ecliptic / Zodiac Rete Ring (23.5 deg tilt)
const ringBGeo = new THREE.TorusGeometry(3.2, 0.04, 20, 140);
const ringBMesh = new THREE.Mesh(ringBGeo, antiqueBrassMat);
ringBMesh.rotation.x = Math.PI / 7.5;

// Clockwork Spur Gear Train
function createSpurGear(radius, toothCount, thickness, material) {
  const gear = new THREE.Group();
  gear.add(new THREE.Mesh(new THREE.TorusGeometry(radius, thickness, 16, 64), material));
  for (let i = 0; i < toothCount; i++) {
    const angle = (Math.PI * 2 / toothCount) * i;
    const tooth = new THREE.Mesh(new THREE.BoxGeometry(0.12, 0.22, thickness * 2), material);
    tooth.position.set(Math.cos(angle) * (radius + 0.08), Math.sin(angle) * (radius + 0.08), 0);
    tooth.rotation.z = angle;
    gear.add(tooth);
  }
  return gear;
}

// Gothic Cathedral Buttress Arches
for (let i = 0; i < 6; i++) {
  const angle = (Math.PI * 2 / 6) * i;
  const ribCurve = new THREE.QuadraticBezierCurve3(
    new THREE.Vector3(Math.cos(angle) * 3.8, -2.6, Math.sin(angle) * 3.8),
    new THREE.Vector3(Math.cos(angle) * 4.4, 0, Math.sin(angle) * 4.4),
    new THREE.Vector3(Math.cos(angle) * 1.5, 2.8, Math.sin(angle) * 1.5)
  );
  gothicPillarsGroup.add(new THREE.Mesh(new THREE.TubeGeometry(ribCurve, 32, 0.05, 8, false), antiqueBrassMat));
}
```
